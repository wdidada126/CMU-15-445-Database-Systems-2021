46 AggregateValue GenerateInitialAggregateValue() {
47 std::vector<Value> values{};
48 for (const auto &agg_type : agg_types_) {
49 switch (agg_type) {
50 case AggregationType::CountAggregate:
51 // Count starts at zero.
52 values.emplace_back(ValueFactory::GetIntegerValue(0));
53 break;
54 case AggregationType::SumAggregate:
55 // Sum starts at zero.
56 values.emplace_back(ValueFactory::GetIntegerValue(0));
57 break;
58 case AggregationType::MinAggregate:
59 // Min starts at INT_MAX.
60 values.emplace_back(ValueFactory::GetIntegerValue(BUSTUB_INT32_MAX));
61 break;
62 case AggregationType::MaxAggregate:
63 // Max starts at INT_MIN.
64 values.emplace_back(ValueFactory::GetIntegerValue(BUSTUB_INT32_MIN));
65 break;
66 }
67 }
68 return {values};
69 }

76 void CombineAggregateValues(AggregateValue *result, const AggregateValue &input) {
77 for (uint32_t i = 0; i < agg_exprs_.size(); i++) {
78 switch (agg_types_[i]) {
79 case AggregationType::CountAggregate:
80 // Count increases by one.
81 result->aggregates_[i] = result->aggregates_[i].Add(ValueFactory::GetIntegerValue(1));
82 break;
83 case AggregationType::SumAggregate:
84 // Sum increases by addition.
85 result->aggregates_[i] = result->aggregates_[i].Add(input.aggregates_[i]);
86 break;
87 case AggregationType::MinAggregate:
88 // Min is just the min.
89 result->aggregates_[i] = result->aggregates_[i].Min(input.aggregates_[i]);
90 break;
91 case AggregationType::MaxAggregate:
92 // Max is just the max.
93 result->aggregates_[i] = result->aggregates_[i].Max(input.aggregates_[i]);
94 break;
95 }
96 }
97 }

104 void InsertCombine(const AggregateKey &agg_key, const AggregateValue &agg_val) {
105 if (ht_.count(agg_key) == 0) {
106 ht_.insert({agg_key, GenerateInitialAggregateValue()});
107 }
108 CombineAggregateValues(&ht_[agg_key], agg_val);
109 }
