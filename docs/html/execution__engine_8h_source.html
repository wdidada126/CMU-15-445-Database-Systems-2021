59 // Execute the query plan
60 try {
61 Tuple tuple;
62 RID rid;
63 while (executor->Next(&tuple, &rid)) {
64 if (result_set != nullptr && plan_type != PlanType::Update && plan_type != PlanType::Insert &&
65 plan_type != PlanType::Delete) {
66 result_set->emplace_back(tuple);
67 }
68 }
69 } catch (TransactionAbortException &e) {
70 txn_mgr_->Abort(txn);
71 return false;
72 } catch (Exception &e) {
73 txn_mgr_->Abort(txn);
74 return false;
75 }
76
77 if (txn->GetState() == TransactionState::ABORTED) {
78 txn_mgr_->Abort(txn);
79 return false;
80 }
81
82 return true;
83 }
