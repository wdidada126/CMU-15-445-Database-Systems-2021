44 void SetUp() override {
45 ::testing::Test::SetUp();
46
47 // Initialize the database subsystems
48 lock_manager_ = std::make_unique<LockManager>();
49 disk_manager_ = std::make_unique<DiskManager>("executor_test.db");
50 bpm_ = std::make_unique<BufferPoolManagerInstance>(32, disk_manager_.get());
51 txn_mgr_ = std::make_unique<TransactionManager>(lock_manager_.get(), log_manager_.get());
52 catalog_ = std::make_unique<Catalog>(bpm_.get(), lock_manager_.get(), log_manager_.get());
53
54 // Open a transaction for the test
55 txn_ = txn_mgr_->Begin();
56
57 // Create an executor context for our executors
58 exec_ctx_ =
59 std::make_unique<ExecutorContext>(txn_, catalog_.get(), bpm_.get(), txn_mgr_.get(), lock_manager_.get());
60
61 // Generate test tables
62 TableGenerator gen{exec_ctx_.get()};
63 gen.GenerateTestTables();
64
65 // Construct the executor engine for the test
66 execution_engine_ = std::make_unique<ExecutionEngine>(bpm_.get(), txn_mgr_.get(), catalog_.get());
67 }

70 void TearDown() override {
71 // Commit our transaction
72 txn_mgr_->Commit(txn_);
73
74 // Shut down the disk manager and clean up the transaction

203 const Schema *MakeOutputSchema(const std::vector<std::pair<std::string, const AbstractExpression *>> &exprs) {
204 std::vector<Column> cols;
205 cols.reserve(exprs.size());
206 for (const auto &input : exprs) {
207 if (input.second->GetReturnType() != TypeId::VARCHAR) {
208 cols.emplace_back(input.first, input.second->GetReturnType(), input.second);
209 } else {
210 cols.emplace_back(input.first, input.second->GetReturnType(), MAX_VARCHAR_SIZE, input.second);

212 }
