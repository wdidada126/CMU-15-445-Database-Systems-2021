27class HashUtil {
28 private:
29 static const hash_t PRIME_FACTOR = 10000019;
30
31 public:
32 static inline hash_t HashBytes(const char *bytes, size_t length) {
33 // https://github.com/greenplum-db/gpos/blob/b53c1acd6285de94044ff91fbee91589543feba1/libgpos/src/utils.cpp#L126
34 hash_t hash = length;
35 for (size_t i = 0; i < length; ++i) {
36 hash = ((hash << 5) ^ (hash >> 27)) ^ bytes[i];
37 }
38 return hash;
39 }
40
41 static inline hash_t CombineHashes(hash_t l, hash_t r) {
42 hash_t both[2] = {};
43 both[0] = l;
44 both[1] = r;
45 return HashBytes(reinterpret_cast<char *>(both), sizeof(hash_t) * 2);
46 }
47
48 static inline hash_t SumHashes(hash_t l, hash_t r) { return (l % PRIME_FACTOR + r % PRIME_FACTOR) % PRIME_FACTOR; }
49
50 template <typename T>
51 static inline hash_t Hash(const T *ptr) {
52 return HashBytes(reinterpret_cast<const char *>(ptr), sizeof(T));
53 }
54
55 template <typename T>
56 static inline hash_t HashPtr(const T *ptr) {
57 return HashBytes(reinterpret_cast<const char *>(&ptr), sizeof(void *));
58 }
59
61 static inline hash_t HashValue(const Value *val) {
62 switch (val->GetTypeId()) {
63 case TypeId::TINYINT: {
64 auto raw = static_cast<int64_t>(val->GetAs<int8_t>());
65 return Hash<int64_t>(&raw);
66 }
67 case TypeId::SMALLINT: {
68 auto raw = static_cast<int64_t>(val->GetAs<int16_t>());
69 return Hash<int64_t>(&raw);
70 }
71 case TypeId::INTEGER: {
72 auto raw = static_cast<int64_t>(val->GetAs<int32_t>());
73 return Hash<int64_t>(&raw);
74 }
75 case TypeId::BIGINT: {
76 auto raw = static_cast<int64_t>(val->GetAs<int64_t>());
77 return Hash<int64_t>(&raw);
78 }
79 case TypeId::BOOLEAN: {
80 auto raw = val->GetAs<bool>();
81 return Hash<bool>(&raw);
82 }
83 case TypeId::DECIMAL: {
84 auto raw = val->GetAs<double>();
85 return Hash<double>(&raw);
86 }
87 case TypeId::VARCHAR: {
88 auto raw = val->GetData();
89 auto len = val->GetLength();
90 return HashBytes(raw, len);
91 }
92 case TypeId::TIMESTAMP: {
93 auto raw = val->GetAs<uint64_t>();
94 return Hash<uint64_t>(&raw);
95 }
96 default: {
97 BUSTUB_ASSERT(false, "Unsupported type.");
98 }
99 }
100 }
101};
