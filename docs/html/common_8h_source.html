25// use a fixed schema to construct a random tuple
26Tuple ConstructTuple(Schema *schema) {
27 std::vector<Value> values;
28 Value v(TypeId::INVALID);
29
30 auto seed = std::chrono::system_clock::now().time_since_epoch().count();
31
32 std::mt19937 generator(seed); // mt19937 is a standard mersenne_twister_engine
33
34 for (uint32_t i = 0; i < schema->GetColumnCount(); i++) {
35 // get type
36 const auto &col = schema->GetColumn(i);
37 TypeId type = col.GetType();
38 switch (type) {
39 case TypeId::BOOLEAN:
40 v = Value(type, static_cast<int8_t>(generator() % 2));
41 break;
42 case TypeId::TINYINT:
43 v = Value(type, static_cast<int8_t>(generator()) % 1000);
44 break;
45 case TypeId::SMALLINT:
46 v = Value(type, static_cast<int16_t>(generator()) % 1000);
47 break;
48 case TypeId::INTEGER:
49 v = Value(type, static_cast<int32_t>(generator()) % 1000);
50 break;
51 case TypeId::BIGINT:
52 v = Value(type, static_cast<int64_t>(generator()) % 1000);
53 break;
54 case TypeId::VARCHAR: {
55 static const char alphanum[] =
56 "0123456789"
57 "ABCDEFGHIJKLMNOPQRSTUVWXYZ"
58 "abcdefghijklmnopqrstuvwxyz";
59 auto len = static_cast<uint32_t>(1 + generator() % 9);
60 char s[10];
61 for (uint32_t j = 0; j < len; ++j) {
62 s[j] = alphanum[generator() % (sizeof(alphanum) - 1)];
63 }
64 s[len] = 0;
65 v = Value(type, s, len + 1, true);
66 break;
67 }
68 default:
69 break;
70 }
71 values.emplace_back(v);
72 }
73 return Tuple(values, schema);
74}
